db.students.aggregate([
  {
    $group: {
      _id: "$department",
      avgGPA: { $avg: "$gpa" }
    }
  }
])

db.students.aggregate([
  {
    $group: {
      _id: "$courseId",
      totalStudents: { $sum: 1 }
    }
  },
  { $sort: { totalStudents: -1 } },
  { $limit: 5 }
])

db.grades.aggregate([
  {
    $group: {
      _id: "$studentId",
      avgGrade: { $avg: "$marks" },
      totalCourses: { $sum: 1 }
    }
  },
  {
    $project: {
      studentId: "$_id",
      avgGrade: 1,
      totalCourses: 1,
      performance: {
        $cond: {
          if: { $gte: ["$avgGrade", 75] },
          then: "Good",
          else: "Needs Improvement"
        }
      }
    }
  }
])
